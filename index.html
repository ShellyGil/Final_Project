import pandas as pd import numpy as np import matplotlib.pyplot as plt import matplotlib.patches as mpatches from scipy import stats from js import document, alert from pyodide.ffi import create_proxy import io, base64 # Global Storage for Data STORE = { "CFA_L": [], "CFA_R": [], "Carr_L": [], "Carr_R": [] } NAMES = { "CFA_L": [], "CFA_R": [], "Carr_L": [], "Carr_R": [] } async def add_files(key, input_id, badge_id): inp = document.getElementById(input_id) files = inp.files if files.length == 0: return for i in range(files.length): f = files.item(i) raw_name = f.name # Simple parser for the text file outputs text = await f.text() clean = text.replace(',', ' ').replace('\t', ' ') tokens = clean.split() nums = [] for t in tokens: try: nums.append(float(t)) except: pass if nums: # Average if multiple numbers, usually 1 per file STORE[key].append(np.mean(nums)) NAMES[key].append(raw_name) bdg = document.getElementById(badge_id) bdg.innerText = f"{len(STORE[key])}" bdg.classList.add("active") inp.value = "" # Wrapper functions for events async def add_cfa_l(e): await add_files("CFA_L", "an_in_CFA_L", "bdg_CFA_L") async def add_cfa_r(e): await add_files("CFA_R", "an_in_CFA_R", "bdg_CFA_R") async def add_carr_l(e): await add_files("Carr_L", "an_in_Carr_L", "bdg_Carr_L") async def add_carr_r(e): await add_files("Carr_R", "an_in_Carr_R", "bdg_Carr_R") def run_analysis(event): if len(STORE["CFA_L"]) == 0: return alert("No Data Loaded") mode = document.getElementById("an-mode").value title = document.getElementById("an-title").value ylabel = document.getElementById("an-ylabel").value fig = None if mode == "ratio": fig = plot_summary(title, ylabel) else: fig = plot_raw(title, ylabel) # Render to div document.getElementById("an-plot-area").innerHTML = "" display(fig, target="an-plot-area", append=False) def plot_summary(title, ylabel): cfa_r, cfa_l = STORE["CFA_R"], STORE["CFA_L"] carr_r, carr_l = STORE["Carr_R"], STORE["Carr_L"] # Fallback if empty if not cfa_r: cfa_r = [0] if not cfa_l: cfa_l = [0] if not carr_r: carr_r = [0] if not carr_l: carr_l = [0] means = [np.mean(cfa_r), np.mean(cfa_l), np.mean(carr_r), np.mean(carr_l)] sems = [stats.sem(cfa_r), stats.sem(cfa_l), stats.sem(carr_r), stats.sem(carr_l)] fig, ax = plt.subplots(figsize=(8, 5)) colors = ["#ccc", "#e74c3c", "#ccc", "#3498db"] ax.bar([0,1,2.5,3.5], means, yerr=sems, capsize=5, color=colors, edgecolor='black') # Scatter points ax.scatter([0]*len(cfa_r), cfa_r, color='black', alpha=0.5, zorder=3) ax.scatter([1]*len(cfa_l), cfa_l, color='black', alpha=0.5, zorder=3) ax.set_xticks([0.5, 3]) ax.set_xticklabels(["CFA", "Carrageenan"]) ax.set_title(title) ax.set_ylabel(ylabel) ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) # Stats Text try: t, p_cfa = stats.ttest_ind(cfa_r, cfa_l) stats_div = document.getElementById("an-stats-area") stats_div.style.display = "block" stats_div.innerText = f"CFA P-Value: {p_cfa:.5f}" except: pass plt.tight_layout() return fig def plot_raw(title, ylabel): fig, ax = plt.subplots(figsize=(10, 5)) vals = STORE["CFA_L"] ax.bar(range(len(vals)), vals, color='#e74c3c') ax.set_title(title + " (Raw CFA Left)") return fig def main(): document.getElementById("btn_add_CFA_L").addEventListener("click", create_proxy(add_cfa_l)) document.getElementById("btn_add_CFA_R").addEventListener("click", create_proxy(add_cfa_r)) document.getElementById("btn_add_Carr_L").addEventListener("click", create_proxy(add_carr_l)) document.getElementById("btn_add_Carr_R").addEventListener("click", create_proxy(add_carr_r)) document.getElementById("an-btn-run").addEventListener("click", create_proxy(run_analysis)) main()
